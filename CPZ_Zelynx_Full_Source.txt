local function applyTheme(name)
    local new = Themes[name]
    if not new then return end
    local old = {}
    for _, k in ipairs(THEME_KEYS) do old[k] = Theme[k] end
    for _, k in ipairs(THEME_KEYS) do Theme[k] = new[k] end
    ACCENT_GRADIENT = makeGradient()
    for _, list in ipairs({ spinners, lineGrads }) do
        for _, g in ipairs(list) do
            if g and g.Parent then g.Color = ACCENT_GRADIENT end
        end
    end
    for _, d in ipairs(ScreenGui:GetDescendants()) do
        for _, prop in ipairs(COLOR_PROPS) do
            local ok, cur = pcall(function() return d[prop] end)
            if ok and typeof(cur) == loadstring(base64decode("Q29sb3Iz"))() then
                for _, k in ipairs(THEME_KEYS) do
                    if cur == old[k] then
                        pcall(function() d[prop] = new[k] end)
                        break
                    end
                end
            end
        end
    end
end

local function makeDraggable(handle, target, onTap)
    local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
    local moved = false
    track(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))
    track(handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))
    track(UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then
                moved = true
                target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end
    end))
    track(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                if not moved and onTap then onTap() end
            end
        end
    end))
end

local function registerSearchItem(frame, text)
    table.insert(SearchableItems, { Frame = frame, Key = string.lower(tostring(text or loadstring(base64decode(""))())) })
end

SearchBox:GetPropertyChangedSignal(loadstring(base64decode("VGV4dA=="))()):Connect(function()
    local q = string.lower(SearchBox.Text or loadstring(base64decode(""))()):gsub(loadstring(base64decode("XiVzKw=="))(), loadstring(base64decode(""))()):gsub(loadstring(base64decode("JXMrJA=="))(), loadstring(base64decode(""))())
    for _, item in ipairs(SearchableItems) do
        if item.Frame and item.Frame.Parent then
            if q == loadstring(base64decode(""))() or string.find(item.Key, q, 1, true) then
                item.Frame.Visible = true
            else
                item.Frame.Visible = false
            end
        end
    end
end)

local function hover(btn, baseT, hoverT, isActive)
    btn.MouseEnter:Connect(function()
        if isActive and isActive() then return end
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundTransparency = hoverT }):Play()
    end)
    btn.MouseLeave:Connect(function()
        if isActive and isActive() then return end
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundTransparency = baseT }):Play()
    end)
end

EmbeddedModules.Gojo = function(Tab)
    local infREnabled = false
    local rTechEnabled = false
    local isLapseDisabled = false
    local gojoKillAllRunning = false
    local tpLoopEnabled = false

    local function getHRP(model)
        if not model then return nil end
        return model:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or model:FindFirstChild(loadstring(base64decode("VG9yc28="))())
    end

    local function getNearestTargetHRP(maxDist)
        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        if not myHRP then return nil end
        local nearestHRP = nil
        local shortestDist = maxDist or 80

        local function checkModel(model)
            if not model or not model:IsA(loadstring(base64decode("TW9kZWw="))()) or model == myChar then return end
            local hum = model:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
            local hrp = getHRP(model)
            if hum and hrp and hum.Health > 0 then
                local dist = (hrp.Position - myHRP.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearestHRP = hrp
                end
            end
        end

        local charFolder = Workspace:FindFirstChild(loadstring(base64decode("Q2hhcmFjdGVycw=="))()) or Workspace
        for _, obj in ipairs(charFolder:GetChildren()) do
            checkModel(obj)
        end
        return nearestHRP
    end

    local function getAllTargetHRPs()
        local myChar = LocalPlayer.Character
        local targets = {}
        local function checkModel(model)
            if not model or not model:IsA(loadstring(base64decode("TW9kZWw="))()) or model == myChar then return end
            local hum = model:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
            local hrp = getHRP(model)
            if hum and hrp and hum.Health > 0 then
                table.insert(targets, { HRP = hrp, Name = model.Name })
            end
        end
        local charFolder = Workspace:FindFirstChild(loadstring(base64decode("Q2hhcmFjdGVycw=="))()) or Workspace
        for _, obj in ipairs(charFolder:GetChildren()) do checkModel(obj) end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then checkModel(p.Character) end
        end
        table.sort(targets, function(a, b) return a.Name < b.Name end)
        local sortedHRPs = {}
        for _, item in ipairs(targets) do table.insert(sortedHRPs, item.HRP) end
        return sortedHRPs
    end

    local function playAnim(animId, duration)
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        if not hum then return nil end
        local animator = hum:FindFirstChildOfClass(loadstring(base64decode("QW5pbWF0b3I="))()) or hum
        local anim = Instance.new(loadstring(base64decode("QW5pbWF0aW9u"))())
        anim.AnimationId = loadstring(base64decode("cmJ4YXNzZXRpZDovLw=="))() .. tostring(animId)
        local track = animator:LoadAnimation(anim)
        track:Play()
        if duration then
            task.delay(duration, function()
                if track and track.IsPlaying then track:Stop() end
            end)
        end
        return track
    end

    local function waitForAnimTrack(animId, timeout)
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        if not hum then return nil end
        local animator = hum:FindFirstChildOfClass(loadstring(base64decode("QW5pbWF0b3I="))()) or hum
        local startTime = tick()
        while tick() - startTime < (timeout or 6) do
            for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                if track.Animation and string.find(tostring(track.Animation.AnimationId), tostring(animId)) then
                    return track
                end
            end
            task.wait(0.05)
        end
        return nil
    end

    local function runGojoKillAll()
        if gojoKillAllRunning then return end
        gojoKillAllRunning = true
        task.spawn(function()
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.GojoService.RE.Ultimate:FireServer()
            end)
            task.wait(0.6)
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.GojoService.RE.RightActivated:FireServer(nil)
            end)
            local targetTrack = waitForAnimTrack(loadstring(base64decode("MTE2ODExODQ2NzE1NDYy"))(), 6)
            if targetTrack then
                task.wait(2)
                local tpStartTime = tick()
                local targetIndex = 1
                while (tick() - tpStartTime < 15.5) and targetTrack and targetTrack.IsPlaying do
                    local allTargets = getAllTargetHRPs()
                    if #allTargets > 0 then
                        if targetIndex > #allTargets then targetIndex = 1 end
                        local currentTargetHRP = allTargets[targetIndex]
                        local myHRP = getHRP(LocalPlayer.Character)
                        if myHRP and currentTargetHRP and currentTargetHRP.Parent then
                            myHRP.CFrame = currentTargetHRP.CFrame
                        end
                        targetIndex = targetIndex + 1
                    end
                    RunService.RenderStepped:Wait()
                end
            end
            gojoKillAllRunning = false
        end)
    end

    if not _G.LarpGojoHooked then
        _G.LarpGojoHooked = true
        pcall(function()

                            isLapseDisabled = true
                            task.delay(3, function() isLapseDisabled = false end)
                        end
                        if self.Name == loadstring(base64decode("UmlnaHRBY3RpdmF0ZWQ="))() and self.Parent and self.Parent.Name == loadstring(base64decode("UkU="))() and self.Parent.Parent and self.Parent.Parent.Name == loadstring(base64decode("R29qb1NlcnZpY2U="))() then
                            if not isLapseDisabled then
                                local passedTarget = args[1]
                                if infREnabled then
                                    task.spawn(function()
                                        playAnim(99920923658527, 0.8)
                                        task.wait(0.8)
                                        local myHRP = getHRP(LocalPlayer.Character)
                                        local targetHRP = getHRP(passedTarget) or getNearestTargetHRP(80)
                                        if myHRP and targetHRP and (targetHRP.Position - myHRP.Position).Magnitude <= 80 then
                                            myHRP.CFrame = targetHRP.CFrame * CFrame.new(0, 0, 3)

                                                    return
                                                end
                                                local lookDir = targetHRP.CFrame.LookVector
                                                myHRP.CFrame = CFrame.new(myHRP.Position, myHRP.Position + lookDir)

    local function toggleACBypass(state)
        tpLoopEnabled = state
        if tpLoopEnabled then
            task.spawn(function()
                while tpLoopEnabled do
                    local char = LocalPlayer.Character
                    local hrp = getHRP(char)
                    if not hrp then task.wait(0.5) continue end
                    local targetUpY = hrp.Position.Y + 800
                    hrp.CFrame = hrp.CFrame + Vector3.new(0, 800, 0)
                    task.wait(0.8)
                    if not tpLoopEnabled then break end
                    char = LocalPlayer.Character
                    hrp = getHRP(char)
                    if not hrp then break end
                    if (targetUpY - hrp.Position.Y) > 200 then
                        pcall(function()
                            ReplicatedStorage.Knit.Knit.Services.JoinService.RE.Reset:FireServer()
                        end)
                        task.wait(1)
                    else
                        tpLoopEnabled = false
                        break
                    end
                end
            end)
        end
    end

EmbeddedModules.Items = function(ItemsTab)
    local RocketSettings = {
        AutoRocketRide = false,
        AutoPerfectAim = false,
        AimSmoothness = 0.25,
        MaxAimDistance = 350
    }

    local ItemSettings = {
        Selected = loadstring(base64decode("Tm9uZQ=="))(),
        ESPEnabled = false,
        MaxDistance = 500,
        ESPColor = Color3.fromRGB(255, 200, 60),
        Highlights = {},
        Billboards = {}
    }

    local State = {
        ActiveRocket = nil,
        ActiveGyro = nil
    }

    local function getBulletsFolder()
        local b = Workspace:FindFirstChild(loadstring(base64decode("QnVsbGV0cw=="))())
        if not b then
            for _, obj in ipairs(Workspace:GetChildren()) do
                if string.lower(obj.Name) == loadstring(base64decode("YnVsbGV0cw=="))() then return obj end
            end
        end
        return b
    end

    local function getRocketPart(rocket)
        if not rocket then return nil end
        if rocket:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) then return rocket end
        if rocket:IsA(loadstring(base64decode("TW9kZWw="))()) then
            return rocket.PrimaryPart or rocket:FindFirstChildWhichIsA(loadstring(base64decode("QmFzZVBhcnQ="))())
        end
        return nil
    end

    local function getItemsFolder()
        return Workspace:FindFirstChild(loadstring(base64decode("SXRlbXM="))()) or Workspace:FindFirstChild(loadstring(base64decode("RHJvcHBlZEl0ZW1z"))())
    end

    local function getItemsList()
        local folder = getItemsFolder()
        local list = {}
        if not folder then return { loadstring(base64decode("Tm9uZQ=="))() } end
        for _, item in ipairs(folder:GetChildren()) do
            if (item:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) or item:IsA(loadstring(base64decode("TW9kZWw="))())) and item:FindFirstChildWhichIsA(loadstring(base64decode("UHJveGltaXR5UHJvbXB0"))(), true) then
                if not table.find(list, item.Name) then
                    table.insert(list, item.Name)
                end
            end
        end
        if #list == 0 then table.insert(list, loadstring(base64decode("Tm9uZQ=="))()) end
        table.sort(list)
        return list
    end

    local function grabItem(name)
        if not name or name == loadstring(base64decode("Tm9uZQ=="))() or typeof(fireproximityprompt) ~= loadstring(base64decode("ZnVuY3Rpb24="))() then return false end
        local folder = getItemsFolder()
        local itm = folder and folder:FindFirstChild(name)
        if not itm then return false end
        local pp = itm:FindFirstChildWhichIsA(loadstring(base64decode("UHJveGltaXR5UHJvbXB0"))(), true)
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        if not (pp and hrp) then return false end
        local targetCFrame = itm:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) and itm.CFrame or (itm.PrimaryPart and itm.PrimaryPart.CFrame or itm:GetPivot())
        task.spawn(function()
            for _ = 1, 3 do
                if not (itm.Parent and hrp.Parent) then break end
                hrp.CFrame = targetCFrame + Vector3.new(0, 1, 0)
                pcall(fireproximityprompt, pp, 0)
                task.wait(0.15)
            end
        end)
        return true
    end

    local function clearItemESP()
        for _, hl in pairs(ItemSettings.Highlights) do
            if hl and hl.Parent then hl:Destroy() end
        end
        for _, bb in pairs(ItemSettings.Billboards) do
            if bb and bb.Parent then bb:Destroy() end
        end
        table.clear(ItemSettings.Highlights)
        table.clear(ItemSettings.Billboards)
    end

    local function updateItemESP()
        if not ItemSettings.ESPEnabled then clearItemESP() return end
        local folder = getItemsFolder()
        if not folder then return end
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        local myPos = hrp and hrp.Position or Vector3.zero

        for _, item in ipairs(folder:GetChildren()) do
            local part = item:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) and item or (item:IsA(loadstring(base64decode("TW9kZWw="))()) and (item.PrimaryPart or item:FindFirstChildWhichIsA(loadstring(base64decode("QmFzZVBhcnQ="))())))
            if part then
                local dist = (part.Position - myPos).Magnitude
                if dist <= ItemSettings.MaxDistance then
                    local hl = ItemSettings.Highlights[item]
                    if not (hl and hl.Parent) then
                        hl = Instance.new(loadstring(base64decode("SGlnaGxpZ2h0"))())
                        hl.Name = loadstring(base64decode("SXRlbUNoYXJt"))()
                        hl.Adornee = item
                        hl.FillColor = ItemSettings.ESPColor
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.5
                        hl.OutlineTransparency = 0.2
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Parent = item
                        ItemSettings.Highlights[item] = hl
                    else
                        hl.FillColor = ItemSettings.ESPColor
                    end

                    local bb = ItemSettings.Billboards[item]
                    if not (bb and bb.Parent) then
                        bb = Instance.new(loadstring(base64decode("QmlsbGJvYXJkR3Vp"))())
                        bb.Name = loadstring(base64decode("SXRlbUJpbGxib2FyZA=="))()
                        bb.Adornee = part
                        bb.Size = UDim2.new(0, 120, 0, 30)
                        bb.StudsOffset = Vector3.new(0, 2, 0)
                        bb.AlwaysOnTop = true
                        bb.Parent = item

                        local lbl = Instance.new(loadstring(base64decode("VGV4dExhYmVs"))(), bb)
                        lbl.Name = loadstring(base64decode("TGFiZWw="))()
                        lbl.Size = UDim2.new(1, 0, 1, 0)
                        lbl.BackgroundTransparency = 1
                        lbl.TextColor3 = ItemSettings.ESPColor
                        lbl.TextStrokeTransparency = 0
                        lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        lbl.Font = Enum.Font.SourceSansBold
                        lbl.TextSize = 13
                        lbl.Text = string.format(loadstring(base64decode("JXMgWyVkbV0="))(), item.Name, math.floor(dist))
                        ItemSettings.Billboards[item] = bb
                    else
                        local lbl = bb:FindFirstChild(loadstring(base64decode("TGFiZWw="))())
                        if lbl then
                            lbl.TextColor3 = ItemSettings.ESPColor
                            lbl.Text = string.format(loadstring(base64decode("JXMgWyVkbV0="))(), item.Name, math.floor(dist))
                        end
                    end
                else
                    if ItemSettings.Highlights[item] then ItemSettings.Highlights[item]:Destroy() ItemSettings.Highlights[item] = nil end
                    if ItemSettings.Billboards[item] then ItemSettings.Billboards[item]:Destroy() ItemSettings.Billboards[item] = nil end
                end
            end
        end
    end

    local function getNearestEnemy(maxDist)
        local shortest = maxDist or RocketSettings.MaxAimDistance
        local nearest = nil
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        if not hrp then return nil end
        local search = Workspace:FindFirstChild(loadstring(base64decode("Q2hhcmFjdGVycw=="))()) or Workspace
        for _, obj in ipairs(search:GetChildren()) do
            if obj:IsA(loadstring(base64decode("TW9kZWw="))()) and obj ~= char then
                local hum = obj:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
                local eHrp = obj:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or obj:FindFirstChild(loadstring(base64decode("VG9yc28="))())
                if hum and eHrp and hum.Health > 0 and not obj:FindFirstChildOfClass(loadstring(base64decode("Rm9yY2VGaWVsZA=="))()) then
                    local d = (eHrp.Position - hrp.Position).Magnitude
                    if d <= shortest then
                        shortest = d
                        nearest = obj
                    end
                end
            end
        end
        return nearest
    end

    local function attachGyroToRocket(rocket)
        local rPart = getRocketPart(rocket)
        if not rPart then return nil end
        local existing = rPart:FindFirstChild(loadstring(base64decode("Um9ja2V0U3RlZXJHeXJv"))())
        if existing and existing:IsA(loadstring(base64decode("Qm9keUd5cm8="))()) then return existing end
        local bg = Instance.new(loadstring(base64decode("Qm9keUd5cm8="))())
        bg.Name = loadstring(base64decode("Um9ja2V0U3RlZXJHeXJv"))()
        bg.MaxTorque = Vector3.new(400000, 400000, 400000)
        bg.P = 15000
        bg.D = 400
        bg.CFrame = rPart.CFrame
        bg.Parent = rPart
        return bg
    end

    local function handleRocketAdded(child)
        if not (child.Name == loadstring(base64decode("Um9ja2V0"))() or string.lower(child.Name):find(loadstring(base64decode("cm9ja2V0"))())) then return end
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        if not hrp then return end
        task.wait(0.03)
        local rPart = getRocketPart(child)
        if rPart and (rPart.Position - hrp.Position).Magnitude <= 28 then
            State.ActiveRocket = child
            State.ActiveGyro = attachGyroToRocket(child)
            if RocketSettings.AutoRocketRide then
                hrp.CFrame = rPart.CFrame * CFrame.new(0, 3.2, 0)
                hrp.AssemblyLinearVelocity = rPart.AssemblyLinearVelocity
            end
        end
    end

    local bullets = getBulletsFolder()
    if bullets then
        track(bullets.ChildAdded:Connect(handleRocketAdded))
    end
    track(Workspace.ChildAdded:Connect(function(c)
        if string.lower(c.Name) == loadstring(base64decode("YnVsbGV0cw=="))() then
            track(c.ChildAdded:Connect(handleRocketAdded))
        elseif string.lower(c.Name):find(loadstring(base64decode("cm9ja2V0"))()) then
            handleRocketAdded(c)
        end
    end))

    track(RunService.Heartbeat:Connect(function()
        local rocket = State.ActiveRocket
        if not (rocket and rocket.Parent) then
            if State.ActiveGyro and State.ActiveGyro.Parent then
                pcall(function() State.ActiveGyro:Destroy() end)
            end
            State.ActiveRocket = nil
            State.ActiveGyro = nil
            return
        end
        local rPart = getRocketPart(rocket)
        if not rPart then return end
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        local hum = char and char:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())

            hrp.CFrame = rPart.CFrame * CFrame.new(0, 3.2, 0)
            hrp.AssemblyLinearVelocity = rPart.AssemblyLinearVelocity
            hrp.AssemblyAngularVelocity = Vector3.zero
        end
        if State.ActiveGyro then
            if RocketSettings.AutoPerfectAim then
                local target = getNearestEnemy(RocketSettings.MaxAimDistance)
                local eHrp = target and (target:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or target:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
                if eHrp then
                    local targetLook = CFrame.lookAt(rPart.Position, eHrp.Position)
                    State.ActiveGyro.CFrame = State.ActiveGyro.CFrame:Lerp(targetLook, RocketSettings.AimSmoothness)
                end
            elseif Camera then
                local camLook = Camera.CFrame.LookVector
                State.ActiveGyro.CFrame = State.ActiveGyro.CFrame:Lerp(CFrame.lookAt(rPart.Position, rPart.Position + camLook), 0.3)
            end
        end

            if ActivatedRemote then pcall(function() ActivatedRemote:FireServer(nil) end) end

        GlobalBlockCount = GlobalBlockCount + 1

        if GlobalBlockCount == 0 and IsPhysicallyBlocking then
            IsPhysicallyBlocking = false

            if DeactivatedRemote then pcall(function() DeactivatedRemote:FireServer() end) end
            if sourceChar and not isTargetInvalid(sourceChar) then
                ExecutePunish(sourceChar)

            end
        end)

    local function GetNearestLockOnTarget()
        local nearest, bestScore = nil, math.huge
        local myChar = LocalPlayer.Character
        local myHRP = getRootPart(myChar)
        if not myHRP or not Camera then return nil end
        local playerPos = myHRP.Position
        local camLook = Camera.CFrame.LookVector

        local function evaluateModel(targetModel)
            local hrp = getRootPart(targetModel)
            local hum = targetModel:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
            if not (hrp and hum and hum.Health > 0) or targetModel:FindFirstChildOfClass(loadstring(base64decode("Rm9yY2VGaWVsZA=="))()) then return end
            local diff = hrp.Position - playerPos
            local dist = diff.Magnitude
            if dist > 350 or dist < 1 then return end
            local score = dist + ((1 - math.clamp(camLook:Dot(diff.Unit), -1, 1)) * 120)
            if score < bestScore then
                bestScore = score
                nearest = targetModel
            end
        end

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then evaluateModel(p.Character) end
        end
        if not nearest then
            local folder = Workspace:FindFirstChild(loadstring(base64decode("TGl2ZQ=="))()) or Workspace:FindFirstChild(loadstring(base64decode("Q2hhcmFjdGVycw=="))()) or Workspace
            for _, obj in ipairs(folder:GetChildren()) do
                if obj:IsA(loadstring(base64decode("TW9kZWw="))()) and obj ~= myChar and not Players:GetPlayerFromCharacter(obj) then
                    evaluateModel(obj)
                end
            end
        end
        return nearest
    end

        local char = LocalPlayer.Character
        local hrp = getRootPart(char)
        local hum = char and char:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        if not char or not hrp or not hum or hum.Health <= 0 then cleanupBodyLock(hum) return end
        local target = Settings.LockOnTarget
        local targetHrp = target and getRootPart(target)
        local targetHum = target and target:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        if Settings.LockOn and Settings.BodyLock and not Settings.Fly and targetHrp and targetHum and targetHum.Health > 0 then
            hum.AutoRotate = false
            local gyro = hrp:FindFirstChild(loadstring(base64decode("TG9ja09uX0JvZHlHeXJv"))())
            if not gyro then
                gyro = Instance.new(loadstring(base64decode("Qm9keUd5cm8="))(), hrp)
                gyro.Name = loadstring(base64decode("TG9ja09uX0JvZHlHeXJv"))()

                gyro.P = 35000
                gyro.D = 300

            local tPos = targetHrp.Position
            if Settings.PredictiveLockOn then
                tPos = tPos + (targetHrp.AssemblyLinearVelocity * Settings.PredictionLeadTime)
            end
            gyro.CFrame = CFrame.lookAt(hrp.Position, Vector3.new(tPos.X, hrp.Position.Y, tPos.Z))
        else
            cleanupBodyLock(hum)
        end
    end

            local target = GetNearestLockOnTarget()

                Settings.LockOnTarget = target

                Settings.LockOnTarget = nil

            end

            Settings.LockOnTarget = nil
            cleanupBodyLock(hum)

EmbeddedModules.Event = function(EventTab)

        InstantPP = false,
        BoostReach = false,
        MaxReachDistance = 25,
        DoorESP = false,
        ESPMaxDistance = 5000
    }
    local State = {
        PPConnections = {},
        OriginalHoldDurations = setmetatable({}, { __mode = loadstring(base64decode("aw=="))() }),
        HighlightedDoors = {},
        IsGrabbingAll = false
    }

    local function getRootPart(char)
        char = char or LocalPlayer.Character
        return char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()) or char.PrimaryPart)
    end

    local function isDoorModel(obj)
        if not obj then return false end
        local name = string.lower(obj.Name)
        return name == loadstring(base64decode("aGFsbG93ZWVuZG9vcg=="))() or string.find(name, loadstring(base64decode("aGFsbG93ZWVuZG9vcg=="))(), 1, true) ~= nil
    end

    local function applyInstantPrompt(prompt)
        if not prompt or not prompt:IsA(loadstring(base64decode("UHJveGltaXR5UHJvbXB0"))()) then return end
        pcall(function()
            if State.OriginalHoldDurations[prompt] == nil then
                State.OriginalHoldDurations[prompt] = prompt.HoldDuration
            end
            if Settings.InstantPP then
                prompt.HoldDuration = 0
                prompt.RequiresLineOfSight = false
                if Settings.BoostReach and prompt.MaxActivationDistance < Settings.MaxReachDistance then
                    prompt.MaxActivationDistance = Settings.MaxReachDistance
                end
            end
        end)
    end

    local function setInstantPPState(enabled)
        Settings.InstantPP = enabled
        if enabled then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA(loadstring(base64decode("UHJveGltaXR5UHJvbXB0"))()) then applyInstantPrompt(obj) end
            end
            table.insert(State.PPConnections, Workspace.DescendantAdded:Connect(function(desc)
                if desc:IsA(loadstring(base64decode("UHJveGltaXR5UHJvbXB0"))()) then task.defer(applyInstantPrompt, desc) end
            end))
            table.insert(State.PPConnections, ProximityPromptService.PromptButtonHoldBegan:Connect(function(prompt)
                if not Settings.InstantPP then return end
                pcall(function()
                    prompt.HoldDuration = 0
                    prompt.RequiresLineOfSight = false
                    if fireproximityprompt then fireproximityprompt(prompt) end
                end)
            end))
        else
            for _, c in ipairs(State.PPConnections) do pcall(function() c:Disconnect() end) end
            table.clear(State.PPConnections)
        end
    end

    local function setDoorESPState(enabled)
        Settings.DoorESP = enabled
        if not enabled then
            for door in pairs(State.HighlightedDoors) do
                if door then
                    local hl = door:FindFirstChild(loadstring(base64decode("SGFsbG93ZWVuRG9vcl9IaWdobGlnaHQ="))())
                    if hl then pcall(hl.Destroy, hl) end
                end
            end
            table.clear(State.HighlightedDoors)
            return
        end
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if isDoorModel(obj) and not State.HighlightedDoors[obj] then
                State.HighlightedDoors[obj] = true
                pcall(function()
                    local hl = Instance.new(loadstring(base64decode("SGlnaGxpZ2h0"))(), obj)
                    hl.Name = loadstring(base64decode("SGFsbG93ZWVuRG9vcl9IaWdobGlnaHQ="))()
                    hl.Adornee = obj
                    hl.FillColor = Color3.fromRGB(255, 115, 0)
                    hl.FillTransparency = 0.5
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                end)
            end
        end
    end

    local function getNearestDoor()
        local root = getRootPart()
        if not root then return nil end
        local nearestDoor, shortestDist = nil, math.huge
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if isDoorModel(obj) then
                local part = obj:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) and obj or (obj:IsA(loadstring(base64decode("TW9kZWw="))()) and (obj.PrimaryPart or obj:FindFirstChildWhichIsA(loadstring(base64decode("QmFzZVBhcnQ="))(), true)))
                if part then
                    local d = (part.Position - root.Position).Magnitude
                    if d < shortestDist then
                        shortestDist = d
                        nearestDoor = obj
                    end
                end
            end
        end
        return nearestDoor
    end

    local function teleportToNearestDoor()
        local door = getNearestDoor()
        local root = getRootPart()
        if not door or not root then return end
        local part = door:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) and door or (door:IsA(loadstring(base64decode("TW9kZWw="))()) and (door.PrimaryPart or door:FindFirstChildWhichIsA(loadstring(base64decode("QmFzZVBhcnQ="))(), true)))
        if part then
            root.CFrame = CFrame.lookAt(part.Position + (part.CFrame.LookVector * 4), part.Position)
        end
    end

EmbeddedModules.Ryu = function(RyuTab)

    local noRyuAnimEnabled = false
    local glideInProgress = false
    local Settings = { Time = 0.25, Distance = 3.5, Range = 50, Curve = 12, CamOffset = 4 }

    local function ResetVelocity(hrp)
        if hrp and hrp.Parent then
            pcall(function()
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    local function isTargetInvalid(targetChar)
        if not targetChar or not targetChar.Parent then return true end
        local hum = targetChar:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        local hrp = targetChar:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or targetChar:FindFirstChild(loadstring(base64decode("VG9yc28="))())
        if not hum or not hrp or hum.Health <= 0 then return true end
        if (targetChar:GetAttribute(loadstring(base64decode("UmFnZG9sbA=="))()) or 0) > 0 then return true end
        return false
    end

    local function getNearestTarget()
        local char = LocalPlayer.Character
        local myHRP = char and (char:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or char:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        if not myHRP then return nil end
        local nearest, shortestDist = nil, Settings.Range
        for _, player in ipairs(Players:GetPlayers()) do
            if player ~= LocalPlayer and player.Character and not isTargetInvalid(player.Character) and not player.Character:FindFirstChildOfClass(loadstring(base64decode("Rm9yY2VGaWVsZA=="))()) then
                local hrp = player.Character:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or player.Character:FindFirstChild(loadstring(base64decode("VG9yc28="))())
                if hrp then
                    local dist = (hrp.Position - myHRP.Position).Magnitude
                    if dist < shortestDist then
                        shortestDist = dist
                        nearest = player.Character
                    end
                end
            end
        end
        return nearest
    end

    local function triggerRyuTech()
        if glideInProgress then return end
        local targetChar = getNearestTarget()
        if not targetChar or isTargetInvalid(targetChar) then return end
        local myChar = LocalPlayer.Character
        local myHRP = myChar and (myChar:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or myChar:FindFirstChild(loadstring(base64decode("VG9yc28="))()))
        local myHumanoid = myChar and myChar:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        local tHRP = targetChar:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))()) or targetChar:FindFirstChild(loadstring(base64decode("VG9yc28="))())
        if not (myHRP and myHumanoid and myHumanoid.Health > 0 and tHRP) then return end
        glideInProgress = true
        pcall(function()
            local knit = ReplicatedStorage:FindFirstChild(loadstring(base64decode("S25pdA=="))())
            local services = knit and (knit:FindFirstChild(loadstring(base64decode("S25pdA=="))()) and knit.Knit:FindFirstChild(loadstring(base64decode("U2VydmljZXM="))()) or knit:FindFirstChild(loadstring(base64decode("U2VydmljZXM="))()))
            local svc = services and services:FindFirstChild(loadstring(base64decode("R3Jhbml0ZUJsYXN0U2VydmljZQ=="))())
            local gbMove = myChar:FindFirstChild(loadstring(base64decode("TW92ZXNldA=="))()) and myChar.Moveset:FindFirstChild(loadstring(base64decode("R3Jhbml0ZSBCbGFzdA=="))())
            if svc and gbMove then
                task.delay(0.10, function()
                    pcall(function() svc.RE.Activated:FireServer(gbMove, nil) end)
                    task.delay(0.10, function()
                        pcall(function() svc.RE.Deactivated:FireServer(gbMove, false) end)
                    end)
                end)
            end
        end)
        local startPos = myHRP.Position
        local initialRightVec = Vector3.new(tHRP.CFrame.RightVector.X, 0, tHRP.CFrame.RightVector.Z).Unit
        local sideSign = (startPos - tHRP.Position):Dot(initialRightVec) >= 0 and 1 or -1
        local startTime = os.clock()
        myHumanoid.AutoRotate = false
        local conn
        conn = RunService.Heartbeat:Connect(function()
            local rawAlpha = math.clamp((os.clock() - startTime) / Settings.Time, 0, 1)
            if rawAlpha >= 1 or isTargetInvalid(targetChar) or not (myHRP.Parent and myHumanoid.Health > 0 and tHRP.Parent) then
                conn:Disconnect()
                myHumanoid.AutoRotate = true
                ResetVelocity(myHRP)
                task.delay(0.02, function() glideInProgress = false end)
                return
            end
            local alpha = 1 - math.pow(1 - rawAlpha, 3)
            local currentTargetPos = tHRP.Position
            local targetLook = Vector3.new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z).Unit
            local targetRight = Vector3.new(tHRP.CFrame.RightVector.X, 0, tHRP.CFrame.RightVector.Z).Unit
            local endPos = currentTargetPos - targetLook * Settings.Distance
            local controlAnchor = currentTargetPos + targetRight * (Settings.Curve * sideSign)
            local blendedControl = controlAnchor:Lerp(endPos, alpha)
            local tInv = 1 - alpha
            local curvePos = (tInv * tInv) * startPos + (2 * tInv * alpha) * blendedControl + (alpha * alpha) * endPos
            local currentY = startPos.Y + (endPos.Y - startPos.Y) * alpha
            myHRP.CFrame = CFrame.lookAt(Vector3.new(curvePos.X, currentY, curvePos.Z), Vector3.new(currentTargetPos.X, currentY, currentTargetPos.Z))
            ResetVelocity(myHRP)
        end)
    end

    RyuTab:CreateSection(loadstring(base64decode("Q29tYmF0"))())
    RyuTab:CreateToggle({ Name = loadstring(base64decode("Unl1IFRlY2g="))(), CurrentValue = false, Callback = function(v) ryuTechEnabled = v end })
    RyuTab:CreateToggle({ Name = loadstring(base64decode("Tm8gQW5pbWF0aW9u"))(), CurrentValue = false, Callback = function(v) noRyuAnimEnabled = v end })
    RyuTab:CreateButton({ Name = loadstring(base64decode("VHJpZ2dlciBUZWNo"))(), Callback = triggerRyuTech })
    RyuTab:CreateKeybind({ Name = loadstring(base64decode("VGVjaCBLZXk="))(), CurrentKeybind = loadstring(base64decode("Qw=="))(), Callback = function() if ryuTechEnabled then triggerRyuTech() end end })
    RyuTab:CreateSection(loadstring(base64decode("U2V0dGluZ3M="))())
    RyuTab:CreateSlider({ Name = loadstring(base64decode("UmFuZ2U="))(), Range = {10, 100}, Increment = 5, Suffix = loadstring(base64decode("IHN0dWRz"))(), CurrentValue = 50, Callback = function(v) Settings.Range = v end })
    RyuTab:CreateSlider({ Name = loadstring(base64decode("RGlzdGFuY2U="))(), Range = {1, 10}, Increment = 0.5, Suffix = loadstring(base64decode("IHN0dWRz"))(), CurrentValue = 3.5, Callback = function(v) Settings.Distance = v end })
    RyuTab:CreateSlider({ Name = loadstring(base64decode("VGltZQ=="))(), Range = {0.1, 0.6}, Increment = 0.01, Suffix = loadstring(base64decode("cw=="))(), CurrentValue = 0.25, Callback = function(v) Settings.Time = v end })
    RyuTab:CreateSlider({ Name = loadstring(base64decode("Q3VydmU="))(), Range = {0, 25}, Increment = 1, CurrentValue = 12, Callback = function(v) Settings.Curve = v end })

    local function stopSealTech()
        if state.animRenderConnection then
            state.animRenderConnection:Disconnect()
            state.animRenderConnection = nil
        end
        local joints = getCharacterJoints()
        if joints then
            for jName, j in pairs(joints) do
                if j and state.cachedDefaultC0s[jName] then j.C0 = state.cachedDefaultC0s[jName] end
            end
        end
        table.clear(state.cachedDefaultC0s)
    end

    local function startSealTech()
        stopSealTech()
        local joints = getCharacterJoints()
        if not joints then return end
        for name, j in pairs(joints) do
            if j then state.cachedDefaultC0s[name] = j.C0 end
        end
        state.animRenderConnection = RunService.RenderStepped:Connect(function()
            local cur = getCharacterJoints()
            if not cur then stopSealTech() return end
            if cur[loadstring(base64decode("TGVmdCBTaG91bGRlcg=="))()] and state.cachedDefaultC0s[loadstring(base64decode("TGVmdCBTaG91bGRlcg=="))()] then
                cur[loadstring(base64decode("TGVmdCBTaG91bGRlcg=="))()].C0 = state.cachedDefaultC0s[loadstring(base64decode("TGVmdCBTaG91bGRlcg=="))()] * CFrame.new(0, -58.68, 0)
            end
            if cur[loadstring(base64decode("UmlnaHQgU2hvdWxkZXI="))()] and state.cachedDefaultC0s[loadstring(base64decode("UmlnaHQgU2hvdWxkZXI="))()] then
                cur[loadstring(base64decode("UmlnaHQgU2hvdWxkZXI="))()].C0 = state.cachedDefaultC0s[loadstring(base64decode("UmlnaHQgU2hvdWxkZXI="))()] * CFrame.new(0, -69.16, 0)
            end
        end)
    end

        state.savedElevation = nil
        if state.activeAnimTrack then
            pcall(function() state.activeAnimTrack:AdjustSpeed(1) end)
            state.activeAnimTrack = nil
        end
        if state.frozenTargetChar then
            local tHRP = state.frozenTargetChar:FindFirstChild(loadstring(base64decode("SHVtYW5vaWRSb290UGFydA=="))())
            if tHRP then tHRP.Anchored = false end
        end
        state.frozenTargetChar = nil
    end

    track(RunService.Heartbeat:Connect(function()
        if not state.infStunEnabled then return end
        local myChar = LocalPlayer.Character
        local hum = myChar and myChar:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
        local animator = hum and hum:FindFirstChildOfClass(loadstring(base64decode("QW5pbWF0b3I="))())

                if animId:find(loadstring(base64decode("MTA4NDE4NTU0ODg3NjU2"))()) and trackObj.TimePosition >= 1.81 then
                    trackObj:AdjustSpeed(0)
                    state.activeAnimTrack = trackObj

                    if myHRP then state.savedElevation = myHRP.Position.Y end
                    state.isFrozen = true

                        if state.isFrozen then unfreezeBoth() end

            end
        end
    end))

    local function findNewClosestTarget()
        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        if not myHRP then return nil end
        local maxRangeSq = Settings.MaxTargetRange ^ 2
        local myPos = myHRP.Position

        if Settings.TargetMode == loadstring(base64decode("U2VsZWN0ZWQ="))() then
            local pName = Settings.SelectedPlayerName
            if pName and pName ~= loadstring(base64decode("Tm9uZQ=="))() then
                local targetPlayer = Players:FindFirstChild(pName)
                if targetPlayer and targetPlayer.Character and isTargetValidAndAlive(targetPlayer.Character) then
                    return targetPlayer.Character
                end
            end
            return nil
        end

        local best, shortest = nil, maxRangeSq
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and isTargetValidAndAlive(p.Character) then
                local tHRP = getHRP(p.Character)
                local dSq = (tHRP.Position - myPos).Magnitude ^ 2
                if dSq < shortest then
                    shortest = dSq
                    best = p.Character
                end
            end
        end

        if not best then
            local charsFolder = Workspace:FindFirstChild(loadstring(base64decode("Q2hhcmFjdGVycw=="))()) or Workspace:FindFirstChild(loadstring(base64decode("TGl2ZQ=="))())
            local pool = charsFolder and charsFolder:GetChildren() or Workspace:GetChildren()
            for _, obj in ipairs(pool) do
                if obj:IsA(loadstring(base64decode("TW9kZWw="))()) and obj ~= myChar and not Players:GetPlayerFromCharacter(obj) and isTargetValidAndAlive(obj) then
                    local tHRP = getHRP(obj)
                    local dSq = (tHRP.Position - myPos).Magnitude ^ 2
                    if dSq < shortest then
                        shortest = dSq
                        best = obj
                    end
                end
            end
        end
        return best

    local function stepOrbitFrame(deltaTime)
        local anyActive = Settings.OrbitEnabled or Settings.LockBehindEnabled
        if not anyActive then
            if State.WasOrbiting then
                State.WasOrbiting = false
                State.LockedTarget = nil
                restoreCameraSubject()
                local myChar = LocalPlayer.Character
                local myHum = myChar and myChar:FindFirstChildOfClass(loadstring(base64decode("SHVtYW5vaWQ="))())
                if myHum then myHum.AutoRotate = true end
            end
            return
        end

        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        local myHum = getAliveHumanoid(myChar)
        if not myHRP or not myHum then
            restoreCameraSubject()
            return
        end

        local targetChar = getLockedOrClosestTarget()
        local tHRP = getHRP(targetChar)
        local tHum = getAliveHumanoid(targetChar)
        if not (tHRP and tHum) then
            restoreCameraSubject()
            if not myHum.AutoRotate then myHum.AutoRotate = true end
            return
        end

        State.WasOrbiting = true

        if (Settings.AutoNoclip or Settings.OrbitEnabled) and myChar then
            for _, part in ipairs(myChar:GetDescendants()) do
                if part:IsA(loadstring(base64decode("QmFzZVBhcnQ="))()) and part.CanCollide then part.CanCollide = false end
            end
        end

        if Settings.LockBehindEnabled then
            restoreCameraSubject()
            myHum.AutoRotate = false
            local tPos = tHRP.Position
            local tLook = Vector3.new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z)
            tLook = (tLook.Magnitude > 0.01) and tLook.Unit or -Vector3.zAxis
            local leadOffset = Settings.PredictMovement and (Vector3.new(tHRP.AssemblyLinearVelocity.X, 0, tHRP.AssemblyLinearVelocity.Z) * math.clamp(deltaTime * 1.5, 0.01, 0.1)) or Vector3.zero
            local desiredPos = tPos - (tLook * Settings.LockBehindDistance) + Vector3.new(0, Settings.LockBehindHeight, 0) + leadOffset
            local goalCFrame = CFrame.lookAt(desiredPos, Vector3.new(tPos.X, desiredPos.Y, tPos.Z))
            myHRP.CFrame = Settings.SmoothTracking and myHRP.CFrame:Lerp(goalCFrame, math.clamp(Settings.TrackingLerpSpeed, 0.05, 1)) or goalCFrame

            if Camera and Camera.CameraSubject ~= tHum then
                Camera.CameraSubject = tHum
            end

            local stepRad = math.rad(Settings.OrbitSpeed) * deltaTime
            if Settings.OrbitSpeed >= 2500 then
                stepRad = stepRad + 2.39996322972865332
            end
            State.CurrentAngle = (State.CurrentAngle + stepRad) % (math.pi * 2)

            local tPos = tHRP.Position
            local calculatedPos = tPos + Vector3.new(
                math.cos(State.CurrentAngle) * Settings.OrbitDistance,
                Settings.OrbitHeight,
                math.sin(State.CurrentAngle) * Settings.OrbitDistance
            )
            myHRP.CFrame = CFrame.lookAt(calculatedPos, Vector3.new(tPos.X, calculatedPos.Y, tPos.Z))
            ResetVelocity(myHRP)

            if Settings.OrbitAutoM1 then
                fireOrbitM1(targetChar)
            end
        end
    end

        if Settings.OrbitEnabled and Settings.OrbitSpeed >= 2500 then

        end

local function hashStr(str)
    local h = 7
    for i = 1, #str do h = (h * 31 + str:byte(i)) % 1000000007 end
    return tostring(h)
end

local function applyUrl(url, fileName)
    if not fsOk then return end
    task.spawn(function()
        local path = BG_DIR .. loadstring(base64decode("Lw=="))() .. fileName
        if not isfile(path) then
            local data = httpGet(url)
            if not data or #data < 200 then return end
            if not pcall(writefile, path, data) then return end
        end
        local ok, asset = pcall(function() return getAsset(path) end)
        if ok and asset and asset ~= loadstring(base64decode(""))() then
            BgImage.Image = asset
        end
    end)
end
