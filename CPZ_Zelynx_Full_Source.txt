local COLOR_PROPS = { "BackgroundColor3", "TextColor3", "ScrollBarImageColor3", "PlaceholderColor3", "Color" }
local function applyTheme(name)
    local new = Themes[name]
    if not new then return end
    local old = {}
    for _, k in ipairs(THEME_KEYS) do old[k] = Theme[k] end
    for _, k in ipairs(THEME_KEYS) do Theme[k] = new[k] end
    ACCENT_GRADIENT = makeGradient()
    for _, list in ipairs({ spinners, lineGrads }) do
        for _, g in ipairs(list) do
            if g and g.Parent then g.Color = ACCENT_GRADIENT end
        end
    end
    for _, d in ipairs(ScreenGui:GetDescendants()) do
        for _, prop in ipairs(COLOR_PROPS) do
            local ok, cur = pcall(function() return d[prop] end)
            if ok and typeof(cur) == "Color3" then
                for _, k in ipairs(THEME_KEYS) do
                    if cur == old[k] then
                        pcall(function() d[prop] = new[k] end)
                        break
                    end
                end
            end
        end
    end
end

local function makeDraggable(handle, target, onTap)
    local dragging, dragInput, dragStart, startPos = false, nil, nil, nil
    local moved = false
    track(handle.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            moved = false
            dragStart = input.Position
            startPos = target.Position
        end
    end))
    track(handle.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end))
    track(UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            if delta.Magnitude > 6 then
                moved = true
                target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            end
        end
    end))
    track(UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            if dragging then
                dragging = false
                if not moved and onTap then onTap() end
            end
        end
    end))
end

    end

local function tr(txt)
    local s = tostring(txt or "")
    return BilingualDict[s] or s
end

local AllTabs = {}
local SearchableItems = {}

local function registerSearchItem(frame, text)
    table.insert(SearchableItems, { Frame = frame, Key = string.lower(tostring(text or "")) })
end

SearchBox:GetPropertyChangedSignal("Text"):Connect(function()
    local q = string.lower(SearchBox.Text or ""):gsub("^%s+", ""):gsub("%s+$", "")
    for _, item in ipairs(SearchableItems) do
        if item.Frame and item.Frame.Parent then
            if q == "" or string.find(item.Key, q, 1, true) then
                item.Frame.Visible = true
            else
                item.Frame.Visible = false
            end
        end
    end
end)

local function hover(btn, baseT, hoverT, isActive)
    btn.MouseEnter:Connect(function()
        if isActive and isActive() then return end
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundTransparency = hoverT }):Play()
    end)
    btn.MouseLeave:Connect(function()
        if isActive and isActive() then return end
        TweenService:Create(btn, TweenInfo.new(0.12), { BackgroundTransparency = baseT }):Play()
    end)
end

    function Tab:CreateSlider(c, optArg)
        local cfg = type(c) == "table" and c or (type(optArg) == "table" and optArg or {})
        local rawName = cfg.Name or cfg.Title or cfg.Text or (type(c) == "string" and c) or "Slider"
        local sName = tr(rawName)
        local range = cfg.Range or { cfg.Min or 0, cfg.Max or 100 }
        local minVal, maxVal = range[1], range[2]
        local inc = cfg.Increment or cfg.Step or (cfg.Rounding == 2 and 0.01) or (cfg.Rounding == 1 and 0.1) or 1
        local suffix = cfg.Suffix and (" " .. cfg.Suffix) or ""
        local curVal = cfg.CurrentValue or cfg.Default or minVal
        local cb = cfg.Callback

        local Card = Instance.new("Frame", Page)
        Card.Size = UDim2.new(1, 0, 0, 50)
        Card.BackgroundColor3 = Theme.Element
        Card.BackgroundTransparency = 0.12
        Card.ZIndex = 3

        local st = Instance.new("UIStroke", Card)
        st.Thickness = 1
        st.Color = Theme.Stroke

        local Lbl = Instance.new("TextLabel", Card)
        Lbl.Size = UDim2.new(1, -90, 0, 20)
        Lbl.Position = UDim2.fromOffset(12, 5)
        Lbl.BackgroundTransparency = 1
        Lbl.Font = Enum.Font.GothamSemibold
        Lbl.Text = sName
        Lbl.TextColor3 = Theme.Text
        Lbl.TextSize = 11
        Lbl.TextXAlignment = Enum.TextXAlignment.Left
        Lbl.TextTruncate = Enum.TextTruncate.AtEnd
        Lbl.ZIndex = 3

        local ValLbl = Instance.new("TextLabel", Card)
        ValLbl.Size = UDim2.fromOffset(80, 20)
        ValLbl.Position = UDim2.new(1, -90, 0, 5)
        ValLbl.BackgroundTransparency = 1
        ValLbl.Font = Enum.Font.GothamBold
        ValLbl.Text = tostring(curVal) .. suffix
        ValLbl.TextColor3 = Theme.Accent
        ValLbl.TextSize = 11
        ValLbl.TextXAlignment = Enum.TextXAlignment.Right
        ValLbl.ZIndex = 3

        local BarBg = Instance.new("TextButton", Card)
        BarBg.Size = UDim2.new(1, -24, 0, 8)
        BarBg.Position = UDim2.fromOffset(12, 32)
        BarBg.BackgroundColor3 = Theme.Off
        BarBg.Text = ""
        BarBg.AutoButtonColor = false
        BarBg.ZIndex = 3
        Instance.new("UICorner", BarBg).CornerRadius = UDim.new(1, 0)

        local initAlpha = math.clamp((curVal - minVal) / math.max(0.0001, maxVal - minVal), 0, 1)
        local Fill = Instance.new("Frame", BarBg)
        Fill.Size = UDim2.new(initAlpha, 0, 1, 0)
        Fill.BackgroundColor3 = Theme.Accent
        Fill.BorderSizePixel = 0
        Fill.ZIndex = 4
        Instance.new("UICorner", Fill).CornerRadius = UDim.new(1, 0)

        local sliding = false
        local function updateVal(pos)
            local relX = math.clamp((pos.X - BarBg.AbsolutePosition.X) / math.max(1, BarBg.AbsoluteSize.X), 0, 1)
            local raw = minVal + (maxVal - minVal) * relX
            local snapped = minVal + math.floor(((raw - minVal) / inc) + 0.5) * inc
            snapped = tonumber(string.format("%.2f", math.clamp(snapped, minVal, maxVal)))
            curVal = snapped
            Fill.Size = UDim2.new((curVal - minVal) / math.max(0.0001, maxVal - minVal), 0, 1, 0)
            ValLbl.Text = tostring(curVal) .. suffix
            if cb then task.spawn(cb, curVal) end
        end

        BarBg.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                sliding = true
                updateVal(input.Position)
            end
        end)
        track(UserInputService.InputChanged:Connect(function(input)
            if sliding and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
                updateVal(input.Position)
            end

EmbeddedModules.Gojo = function(Tab)
    local infREnabled = false
    local rTechEnabled = false
    local isLapseDisabled = false
    local gojoKillAllRunning = false
    local tpLoopEnabled = false

    local function getHRP(model)
        if not model then return nil end
        return model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso")
    end

    local function getNearestTargetHRP(maxDist)
        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        if not myHRP then return nil end
        local nearestHRP = nil
        local shortestDist = maxDist or 80

        local function checkModel(model)
            if not model or not model:IsA("Model") or model == myChar then return end
            local hum = model:FindFirstChildOfClass("Humanoid")
            local hrp = getHRP(model)
            if hum and hrp and hum.Health > 0 then
                local dist = (hrp.Position - myHRP.Position).Magnitude
                if dist < shortestDist then
                    shortestDist = dist
                    nearestHRP = hrp
                end
            end
        end

        local charFolder = Workspace:FindFirstChild("Characters") or Workspace
        for _, obj in ipairs(charFolder:GetChildren()) do
            checkModel(obj)
        end
        return nearestHRP
    end

    local function getAllTargetHRPs()
        local myChar = LocalPlayer.Character
        local targets = {}
        local function checkModel(model)
            if not model or not model:IsA("Model") or model == myChar then return end
            local hum = model:FindFirstChildOfClass("Humanoid")
            local hrp = getHRP(model)
            if hum and hrp and hum.Health > 0 then
                table.insert(targets, { HRP = hrp, Name = model.Name })
            end
        end
        local charFolder = Workspace:FindFirstChild("Characters") or Workspace
        for _, obj in ipairs(charFolder:GetChildren()) do checkModel(obj) end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then checkModel(p.Character) end
        end
        table.sort(targets, function(a, b) return a.Name < b.Name end)
        local sortedHRPs = {}
        for _, item in ipairs(targets) do table.insert(sortedHRPs, item.HRP) end
        return sortedHRPs
    end

    local function playAnim(animId, duration)
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return nil end
        local animator = hum:FindFirstChildOfClass("Animator") or hum
        local anim = Instance.new("Animation")
        anim.AnimationId = "rbxassetid://" .. tostring(animId)
        local track = animator:LoadAnimation(anim)
        track:Play()
        if duration then
            task.delay(duration, function()
                if track and track.IsPlaying then track:Stop() end
            end)
        end
        return track
    end

    local function waitForAnimTrack(animId, timeout)
        local char = LocalPlayer.Character
        if not char then return nil end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return nil end
        local animator = hum:FindFirstChildOfClass("Animator") or hum
        local startTime = tick()
        while tick() - startTime < (timeout or 6) do
            for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
                if track.Animation and string.find(tostring(track.Animation.AnimationId), tostring(animId)) then
                    return track
                end
            end
            task.wait(0.05)
        end
        return nil
    end

    local function runGojoKillAll()
        if gojoKillAllRunning then return end
        gojoKillAllRunning = true
        task.spawn(function()
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.GojoService.RE.Ultimate:FireServer()
            end)
            task.wait(0.6)
            pcall(function()
                ReplicatedStorage.Knit.Knit.Services.GojoService.RE.RightActivated:FireServer(nil)
            end)
            local targetTrack = waitForAnimTrack("116811846715462", 6)
            if targetTrack then
                task.wait(2)
                local tpStartTime = tick()
                local targetIndex = 1
                while (tick() - tpStartTime < 15.5) and targetTrack and targetTrack.IsPlaying do
                    local allTargets = getAllTargetHRPs()
                    if #allTargets > 0 then
                        if targetIndex > #allTargets then targetIndex = 1 end
                        local currentTargetHRP = allTargets[targetIndex]
                        local myHRP = getHRP(LocalPlayer.Character)
                        if myHRP and currentTargetHRP and currentTargetHRP.Parent then
                            myHRP.CFrame = currentTargetHRP.CFrame
                        end
                        targetIndex = targetIndex + 1
                    end
                    RunService.RenderStepped:Wait()
                end
            end
            gojoKillAllRunning = false
        end)
    end

    if not _G.LarpGojoHooked then
        _G.LarpGojoHooked = true
        pcall(function()
            local gmt = getrawmetatable and getrawmetatable(game)
            if gmt then
                local setreadonly = setreadonly or make_writeable or function() end
                setreadonly(gmt, false)
                local oldNamecall = gmt.__namecall
                gmt.__namecall = newcclosure(function(self, ...)
                    local method = getnamecallmethod and getnamecallmethod() or ""

                    if not checkcaller() and method == "FireServer" then

                        end

                            if not isLapseDisabled then
                                local passedTarget = args[1]
                                if infREnabled then
                                    task.spawn(function()
                                        playAnim(99920923658527, 0.8)
                                        task.wait(0.8)
                                        local myHRP = getHRP(LocalPlayer.Character)
                                        local targetHRP = getHRP(passedTarget) or getNearestTargetHRP(80)
                                        if myHRP and targetHRP and (targetHRP.Position - myHRP.Position).Magnitude <= 80 then
                                            myHRP.CFrame = targetHRP.CFrame * CFrame.new(0, 0, 3)

                                                myHRP.CFrame = CFrame.new(myHRP.Position, myHRP.Position + lookDir)
                                                camera.CFrame = CFrame.new(camera.CFrame.Position, camera.CFrame.Position + lookDir)

                                        end

    }

        local b = Workspace:FindFirstChild("Bullets")
        if not b then
            for _, obj in ipairs(Workspace:GetChildren()) do
                if string.lower(obj.Name) == "bullets" then return obj end
            end
        end
        return b
    end

    local function getRocketPart(rocket)
        if not rocket then return nil end
        if rocket:IsA("BasePart") then return rocket end
        if rocket:IsA("Model") then
            return rocket.PrimaryPart or rocket:FindFirstChildWhichIsA("BasePart")
        end
        return nil
    end

    local function getItemsFolder()
        return Workspace:FindFirstChild("Items") or Workspace:FindFirstChild("DroppedItems")
    end

    local function getItemsList()
        local folder = getItemsFolder()
        local list = {}
        if not folder then return { "None" } end
        for _, item in ipairs(folder:GetChildren()) do
            if (item:IsA("BasePart") or item:IsA("Model")) and item:FindFirstChildWhichIsA("ProximityPrompt", true) then
                if not table.find(list, item.Name) then
                    table.insert(list, item.Name)
                end
            end
        end
        if #list == 0 then table.insert(list, "None") end
        table.sort(list)
        return list
    end

    local function grabItem(name)
        if not name or name == "None" or typeof(fireproximityprompt) ~= "function" then return false end
        local folder = getItemsFolder()
        local itm = folder and folder:FindFirstChild(name)
        if not itm then return false end
        local pp = itm:FindFirstChildWhichIsA("ProximityPrompt", true)
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
        if not (pp and hrp) then return false end
        local targetCFrame = itm:IsA("BasePart") and itm.CFrame or (itm.PrimaryPart and itm.PrimaryPart.CFrame or itm:GetPivot())
        task.spawn(function()
            for _ = 1, 3 do
                if not (itm.Parent and hrp.Parent) then break end
                hrp.CFrame = targetCFrame + Vector3.new(0, 1, 0)
                pcall(fireproximityprompt, pp, 0)
                task.wait(0.15)
            end
        end)
        return true
    end

    local function clearItemESP()
        for _, hl in pairs(ItemSettings.Highlights) do
            if hl and hl.Parent then hl:Destroy() end
        end
        for _, bb in pairs(ItemSettings.Billboards) do
            if bb and bb.Parent then bb:Destroy() end
        end
        table.clear(ItemSettings.Highlights)
        table.clear(ItemSettings.Billboards)
    end

    local function updateItemESP()
        if not ItemSettings.ESPEnabled then clearItemESP() return end
        local folder = getItemsFolder()
        if not folder then return end
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
        local myPos = hrp and hrp.Position or Vector3.zero

        for _, item in ipairs(folder:GetChildren()) do
            local part = item:IsA("BasePart") and item or (item:IsA("Model") and (item.PrimaryPart or item:FindFirstChildWhichIsA("BasePart")))
            if part then
                local dist = (part.Position - myPos).Magnitude
                if dist <= ItemSettings.MaxDistance then
                    local hl = ItemSettings.Highlights[item]
                    if not (hl and hl.Parent) then
                        hl = Instance.new("Highlight")
                        hl.Name = "ItemCharm"
                        hl.Adornee = item
                        hl.FillColor = ItemSettings.ESPColor
                        hl.OutlineColor = Color3.fromRGB(255, 255, 255)
                        hl.FillTransparency = 0.5
                        hl.OutlineTransparency = 0.2
                        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                        hl.Parent = item
                        ItemSettings.Highlights[item] = hl
                    else
                        hl.FillColor = ItemSettings.ESPColor
                    end

                    local bb = ItemSettings.Billboards[item]
                    if not (bb and bb.Parent) then
                        bb = Instance.new("BillboardGui")
                        bb.Name = "ItemBillboard"
                        bb.Adornee = part
                        bb.Size = UDim2.new(0, 120, 0, 30)
                        bb.StudsOffset = Vector3.new(0, 2, 0)
                        bb.AlwaysOnTop = true
                        bb.Parent = item

                        local lbl = Instance.new("TextLabel", bb)
                        lbl.Name = "Label"
                        lbl.Size = UDim2.new(1, 0, 1, 0)
                        lbl.BackgroundTransparency = 1
                        lbl.TextColor3 = ItemSettings.ESPColor
                        lbl.TextStrokeTransparency = 0
                        lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        lbl.Font = Enum.Font.SourceSansBold
                        lbl.TextSize = 13
                        lbl.Text = string.format("%s [%dm]", item.Name, math.floor(dist))
                        ItemSettings.Billboards[item] = bb
                    else
                        local lbl = bb:FindFirstChild("Label")
                        if lbl then
                            lbl.TextColor3 = ItemSettings.ESPColor
                            lbl.Text = string.format("%s [%dm]", item.Name, math.floor(dist))
                        end
                    end
                else
                    if ItemSettings.Highlights[item] then ItemSettings.Highlights[item]:Destroy() ItemSettings.Highlights[item] = nil end
                    if ItemSettings.Billboards[item] then ItemSettings.Billboards[item]:Destroy() ItemSettings.Billboards[item] = nil end
                end
            end
        end
    end

    local function getNearestEnemy(maxDist)
        local shortest = maxDist or RocketSettings.MaxAimDistance
        local nearest = nil
        local char = LocalPlayer.Character
        local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
        if not hrp then return nil end
        local search = Workspace:FindFirstChild("Characters") or Workspace
        for _, obj in ipairs(search:GetChildren()) do
            if obj:IsA("Model") and obj ~= char then
                local hum = obj:FindFirstChildOfClass("Humanoid")
                local eHrp = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChild("Torso")
                if hum and eHrp and hum.Health > 0 and not obj:FindFirstChildOfClass("ForceField") then
                    local d = (eHrp.Position - hrp.Position).Magnitude
                    if d <= shortest then
                        shortest = d
                        nearest = obj
                    end
                end
            end
        end
        return nearest
    end

EmbeddedModules.AutoBlock = function(AutoBlockTab)
    local Settings = {
        AutoBlockEnabled = false,
        BlockWhileAttacking = false,
        DetectionRange = 12,
        FacingCheckEnabled = false,
        FacingAngleThreshold = 0.7,
        ProjectileBlock = false,
        ProjectileMaxRange = 30,
        ProjectileFacingSensitivity = 0.3,
        CustomBlockDurationEnabled = false,
        CustomBlockDuration = 0.35,
        PingCompensation = false,
        AutoLockOnBlock = false,
        AutoPunishEnabled = false,
        PunishRange = 25,
        PunishDelay = 0.04,
        NoBlockOnCounter = false,
        NoBlockAndCounterAnime = false,
        HitGlowBlock = false,
        AntiFeintEnabled = false,
    }

    local CounterAnimations = {
        ["135300554211825"] = true, ["135915491872296"] = true,
        ["123414935051274"] = true, ["105376952884290"] = true,
    }
    local CounterAnimeAnimations = {
        ["72933571933445"] = true, ["112519964343154"] = true,
        ["125397194852745"] = true, ["75237089031145"] = true,
        ["104566923474227"] = true, ["82987093810211"] = true,
        ["72828071138653"] = true, ["101956908324027"] = true,
        ["101865783312435"] = true, ["73110085912073"] = true,
        ["99032134144831"] = true,
    }
    local MovementStateKeywords = {
        "run", "chase", "down", "fall", "ragdoll", "idle",
        "walk", "jump", "dash", "climb", "getup", "land",
        "sprint", "movement", "turn", "halt", "hover", "sleep",
        "emote", "spawn", "dance", "wave", "sit", "block"
    }

    local ActiveBlockingTracks = {}
    local ConnectedCharacters = {}
    local GlobalBlockCount = 0
    local IsPhysicallyBlocking = false
    local PunishBusy = false
    local ActivatedRemote, DeactivatedRemote = nil, nil

    local function ResolveBlockRemotes()
        pcall(function()
            local knit = ReplicatedStorage:FindFirstChild("Knit")
            local services = knit and ((knit:FindFirstChild("Knit") and knit.Knit:FindFirstChild("Services")) or knit:FindFirstChild("Services"))
            local blockService = services and services:FindFirstChild("BlockService")
            local re = blockService and blockService:FindFirstChild("RE")
            if re then
                ActivatedRemote = re:FindFirstChild("Activated")
                DeactivatedRemote = re:FindFirstChild("Deactivated")
            end
        end)
    end
    ResolveBlockRemotes()

    local function isTargetInvalid(targetChar)
        if not targetChar or not targetChar.Parent then return true end
        local hum = targetChar:FindFirstChildOfClass("Humanoid")
        local hrp = targetChar:FindFirstChild("HumanoidRootPart") or targetChar:FindFirstChild("Torso")
        if not hum or not hrp or hum.Health <= 0 then return true end
        if (targetChar:GetAttribute("Ragdoll") or 0) > 0 then return true end
        return false
    end

    local function LockFacingTowardThreat(threatPos)
        if not Settings.AutoLockOnBlock or not threatPos then return end
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local flatThreat = Vector3.new(threatPos.X, hrp.Position.Y, threatPos.Z)
        if (flatThreat - hrp.Position).Magnitude > 0.2 then
            hrp.CFrame = CFrame.lookAt(hrp.Position, flatThreat)
        end
    end

    local function ExecutePunish(targetChar)
        if not Settings.AutoPunishEnabled or PunishBusy then return end
        local char = LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local tRoot = targetChar and targetChar:FindFirstChild("HumanoidRootPart")
        if not hrp or not tRoot or isTargetInvalid(targetChar) then return end
        if (tRoot.Position - hrp.Position).Magnitude > Settings.PunishRange then return end
        PunishBusy = true
        task.spawn(function()
            if Settings.PunishDelay > 0 then task.wait(Settings.PunishDelay) end
            if not isTargetInvalid(targetChar) then
                LockFacingTowardThreat(tRoot.Position)
                local ms = char:GetAttribute("Moveset") or LocalPlayer:GetAttribute("Moveset")
                if ms then
                    local cleanName = string.gsub(ms == "Makora" and "Mahoraga" or ms, "%s+", "")
                    local knit = ReplicatedStorage:FindFirstChild("Knit")
                    local services = knit and ((knit:FindFirstChild("Knit") and knit.Knit:FindFirstChild("Services")) or knit:FindFirstChild("Services"))
                    local svc = services and services:FindFirstChild(cleanName .. "Service")
                    local act = svc and svc:FindFirstChild("RE") and svc.RE:FindFirstChild("Activated")
                    if act then pcall(function() act:FireServer(false, targetChar) end) end
                end
            end
            task.wait(0.2)
            PunishBusy = false
        end)
    end

    local function FireBlockActivated(threatPos)
        if threatPos then LockFacingTowardThreat(threatPos) end
        if not IsPhysicallyBlocking then
            IsPhysicallyBlocking = true
            if not ActivatedRemote then ResolveBlockRemotes() end
            if ActivatedRemote then pcall(function() ActivatedRemote:FireServer(nil) end) end
        end
        GlobalBlockCount = GlobalBlockCount + 1
    end

    local function FireBlockDeactivated(sourceChar)
        GlobalBlockCount = math.max(0, GlobalBlockCount - 1)
        if GlobalBlockCount == 0 and IsPhysicallyBlocking then
            IsPhysicallyBlocking = false
            if not DeactivatedRemote then ResolveBlockRemotes() end
            if DeactivatedRemote then pcall(function() DeactivatedRemote:FireServer() end) end
            if sourceChar and not isTargetInvalid(sourceChar) then
                ExecutePunish(sourceChar)
            end
        end
    end

    local function HandleAnimation(character, animationTrack)
        if not Settings.AutoBlockEnabled or not animationTrack or not animationTrack.Animation then return end
        local lowerName = string.lower(animationTrack.Animation.Name or "")
        local rawId = string.match(tostring(animationTrack.Animation.AnimationId), "%d+")
        for i = 1, #MovementStateKeywords do
            if string.find(lowerName, MovementStateKeywords[i], 1, true) then return end
        end
        if Players:GetPlayerFromCharacter(character) == LocalPlayer or isTargetInvalid(character) then return end
        local localChar = LocalPlayer.Character
        local localRoot = localChar and localChar:FindFirstChild("HumanoidRootPart")
        local targetRoot = character:FindFirstChild("HumanoidRootPart")
        if not localRoot or not targetRoot then return end
        if (localRoot.Position - targetRoot.Position).Magnitude > Settings.DetectionRange then return end
        if Settings.FacingCheckEnabled then
            local dirToUs = (localRoot.Position - targetRoot.Position).Unit
            if targetRoot.CFrame.LookVector:Dot(dirToUs) < Settings.FacingAngleThreshold then return end
        end
        local isCounter = string.find(lowerName, "counter", 1, true)
            or (rawId and CounterAnimations[rawId])
            or (rawId and CounterAnimeAnimations[rawId])
        if (Settings.NoBlockOnCounter or Settings.NoBlockAndCounterAnime) and isCounter then return end
        if ActiveBlockingTracks[animationTrack] then return end
        ActiveBlockingTracks[animationTrack] = true
        FireBlockActivated(targetRoot.Position)
        animationTrack.Stopped:Once(function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated(character)
            end
        end)
        task.delay(4, function()
            if ActiveBlockingTracks[animationTrack] then
                ActiveBlockingTracks[animationTrack] = nil
                FireBlockDeactivated(character)
            end
        end)
    end

    local function ConnectCharacter(char)
        if not char or ConnectedCharacters[char] or char == LocalPlayer.Character then return end
        local hum = char:WaitForChild("Humanoid", 4)
        if hum then
            ConnectedCharacters[char] = true
            track(hum.AnimationPlayed:Connect(function(trk)
                HandleAnimation(char, trk)
            end))
        end
    end

    task.spawn(function()
        local chars = Workspace:FindFirstChild("Characters")
        if chars then
            for _, c in ipairs(chars:GetChildren()) do task.spawn(ConnectCharacter, c) end
            track(chars.ChildAdded:Connect(function(c) task.wait(0.2) ConnectCharacter(c) end))
        end
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then task.spawn(ConnectCharacter, p.Character) end
            track(p.CharacterAdded:Connect(function(c) task.wait(0.2) ConnectCharacter(c) end))
        end
        track(Players.PlayerAdded:Connect(function(p)
            track(p.CharacterAdded:Connect(function(c) task.wait(0.2) ConnectCharacter(c) end))
        end))
    end)

    AutoBlockTab:CreateSection("Melee")
    AutoBlockTab:CreateToggle({ Name = "Auto Block", CurrentValue = false, Callback = function(v) Settings.AutoBlockEnabled = v end })
    AutoBlockTab:CreateToggle({ Name = "Block In Attack", CurrentValue = false, Callback = function(v) Settings.BlockWhileAttacking = v end })
    AutoBlockTab:CreateToggle({ Name = "Hit Glow Block", CurrentValue = false, Callback = function(v) Settings.HitGlowBlock = v end })
    AutoBlockTab:CreateToggle({ Name = "Anti Feint", CurrentValue = false, Callback = function(v) Settings.AntiFeintEnabled = v end })
    AutoBlockTab:CreateSlider({ Name = "Range", Range = {5, 50}, Increment = 1, Suffix = " studs", CurrentValue = 12, Callback = function(v) Settings.DetectionRange = v end })
    AutoBlockTab:CreateToggle({ Name = "Facing Check", CurrentValue = false, Callback = function(v) Settings.FacingCheckEnabled = v end })
    AutoBlockTab:CreateSlider({ Name = "Facing Angle", Range = {0, 1}, Increment = 0.05, CurrentValue = 0.7, Callback = function(v) Settings.FacingAngleThreshold = v end })
    AutoBlockTab:CreateSection("Projectile")
    AutoBlockTab:CreateToggle({ Name = "Projectile Block", CurrentValue = false, Callback = function(v) Settings.ProjectileBlock = v end })
    AutoBlockTab:CreateSlider({ Name = "Max Range", Range = {10, 60}, Increment = 1, Suffix = " studs", CurrentValue = 30, Callback = function(v) Settings.ProjectileMaxRange = v end })
    AutoBlockTab:CreateSlider({ Name = "Facing Angle", Range = {0.1, 0.9}, Increment = 0.05, CurrentValue = 0.3, Callback = function(v) Settings.ProjectileFacingSensitivity = v end })
    AutoBlockTab:CreateToggle({ Name = "Custom Duration", CurrentValue = false, Callback = function(v) Settings.CustomBlockDurationEnabled = v end })
    AutoBlockTab:CreateSlider({ Name = "Duration", Range = {0.15, 1.2}, Increment = 0.05, Suffix = "s", CurrentValue = 0.35, Callback = function(v) Settings.CustomBlockDuration = v end })
    AutoBlockTab:CreateToggle({ Name = "Ping Comp", CurrentValue = false, Callback = function(v) Settings.PingCompensation = v end })
    AutoBlockTab:CreateSection("Punish")
    AutoBlockTab:CreateToggle({ Name = "Lock On Block", CurrentValue = false, Callback = function(v) Settings.AutoLockOnBlock = v end })
    AutoBlockTab:CreateToggle({ Name = "Auto Punish", CurrentValue = false, Callback = function(v) Settings.AutoPunishEnabled = v end })

    AutoBlockTab:CreateSlider({ Name = "Punish Delay", Range = {0.0, 0.5}, Increment = 0.01, Suffix = "s", CurrentValue = 0.04, Callback = function(v) Settings.PunishDelay = v end })

    AutoBlockTab:CreateToggle({ Name = "No Block Counter", CurrentValue = false, Callback = function(v) Settings.NoBlockOnCounter = v end })

end

EmbeddedModules.AllChar = function(AllCharTab)
    local Settings = {
        Fly = false,
        FlySpeed = 50,
        Noclip = false,
        DomainWalk = false,
        LockOn = false,
        BodyLock = false,
        MobileGUI = false,
        LockOnKey = Enum.KeyCode.E,
        LockOnTarget = nil,
        PredictiveLockOn = false,
        PredictionLeadTime = 0.18,
        BackflipToggle = false,
        BackflipKey = Enum.KeyCode.Z,
        BackflipHeight = 25,
        BackflipSpeed = 0.45,
        BackflipForward = 10,
        BackflipBackward = 0,
        BackflipRotations = 1.0,
        HitboxExpander = false,
        HitboxSize = 20,
        AutoInstantRespawn = false,
        AutoInstantRespawnHP = 25,
        RagdollWeight = false,
        RagdollGravity = 50,
        AntiCounter = false,
        AntiCounterDist = 12,
        AntiCounterTurn = false,
        AntiCounterStop = true,
        AntiCounterDuration = 0.40
    }

    local State = {
        OriginalGravity = Workspace.Gravity,
        RemoteBlockUntil = 0,
        SuspendedLockOnUntil = 0,
        CounterEnemy = nil,
        DomainOriginals = setmetatable({}, { __mode = "k" }),
        DomainConnections = {},
        CurrentDomainBox = nil,
    }

    local function getRootPart(char)
        char = char or LocalPlayer.Character
        return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso"))
    end

    local function setDomainWalkState(enabled)
        Settings.DomainWalk = enabled
        if not enabled then
            for _, conn in ipairs(State.DomainConnections) do pcall(function() conn:Disconnect() end) end
            table.clear(State.DomainConnections)
            State.CurrentDomainBox = nil
            for part, wasCollidable in pairs(State.DomainOriginals) do
                if part and part.Parent then pcall(function() part.CanCollide = wasCollidable end) end
            end
            table.clear(State.DomainOriginals)
            return
        end
        local function setNoCol(p)
            if State.DomainOriginals[p] == nil and p:IsA("BasePart") then
                State.DomainOriginals[p] = p.CanCollide
                pcall(function() p.CanCollide = false end)
            end
        end
        local function bindFolder(folder)
            if not folder or State.CurrentDomainBox == folder then return end
            State.CurrentDomainBox = folder
            for _, d in ipairs(folder:GetDescendants()) do setNoCol(d) end
            table.insert(State.DomainConnections, folder.DescendantAdded:Connect(function(d)
                if Settings.DomainWalk then setNoCol(d) end
            end))
        end
        table.insert(State.DomainConnections, Workspace.ChildAdded:Connect(function(ch)
            if Settings.DomainWalk and ch.Name == "Domains" then bindFolder(ch) end
        end))
        bindFolder(Workspace:FindFirstChild("Domains"))
    end

    local hitboxHooked = false
    local originalSphereHitbox = nil
    local function HookHitboxController()
        if hitboxHooked then return end
        pcall(function()
            local ps = LocalPlayer:FindFirstChild("PlayerScripts")
            local hbCtrl = ps and ps:FindFirstChild("Controllers") and ps.Controllers:FindFirstChild("Combat") and ps.Controllers.Combat:FindFirstChild("HitboxController")
            if hbCtrl then
                local req = require(hbCtrl)
                if req and type(req.SphereHitbox) == "function" then
                    hitboxHooked = true
                    originalSphereHitbox = req.SphereHitbox
                    req.SphereHitbox = function(self, char, cf, r)
                        if Settings.HitboxExpander then
                            return originalSphereHitbox(self, char, cf, Settings.HitboxSize or r)
                        end
                        return originalSphereHitbox(self, char, cf, r)
                    end
                end
            end
        end)
    end

    local function GetNearestLockOnTarget()
        local nearest, bestScore = nil, math.huge
        local myChar = LocalPlayer.Character
        local myHRP = getRootPart(myChar)
        if not myHRP or not Camera then return nil end
        local playerPos = myHRP.Position
        local camLook = Camera.CFrame.LookVector

        local function evaluateModel(targetModel)
            local hrp = getRootPart(targetModel)
            local hum = targetModel:FindFirstChildOfClass("Humanoid")
            if not (hrp and hum and hum.Health > 0) or targetModel:FindFirstChildOfClass("ForceField") then return end
            local diff = hrp.Position - playerPos
            local dist = diff.Magnitude
            if dist > 350 or dist < 1 then return end
            local score = dist + ((1 - math.clamp(camLook:Dot(diff.Unit), -1, 1)) * 120)
            if score < bestScore then
                bestScore = score
                nearest = targetModel
            end
        end

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character then evaluateModel(p.Character) end
        end
        if not nearest then
            local folder = Workspace:FindFirstChild("Live") or Workspace:FindFirstChild("Characters") or Workspace
            for _, obj in ipairs(folder:GetChildren()) do
                if obj:IsA("Model") and obj ~= myChar and not Players:GetPlayerFromCharacter(obj) then
                    evaluateModel(obj)
                end
            end
        end
        return nearest
    end

    local function cleanupBodyLock(hum)
        local char = LocalPlayer.Character
        if char then
            for _, d in ipairs(char:GetDescendants()) do
                if d:IsA("BodyGyro") and d.Name == "LockOn_BodyGyro" then pcall(d.Destroy, d) end
            end
        end
        if hum and not Settings.Fly then pcall(function() hum.AutoRotate = true end) end
    end

    local function UpdateBodyLock()
        local char = LocalPlayer.Character
        local hrp = getRootPart(char)
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if not char or not hrp or not hum or hum.Health <= 0 then cleanupBodyLock(hum) return end

EmbeddedModules.Event = function(EventTab)
    local Settings = {
        InstantPP = false,
        BoostReach = false,
        MaxReachDistance = 25,
        DoorESP = false,
        ESPMaxDistance = 5000
    }
    local State = {
        PPConnections = {},
        OriginalHoldDurations = setmetatable({}, { __mode = "k" }),
        HighlightedDoors = {},
        IsGrabbingAll = false
    }

    local function getRootPart(char)
        char = char or LocalPlayer.Character
        return char and (char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char.PrimaryPart)
    end

    local function isDoorModel(obj)
        if not obj then return false end
        local name = string.lower(obj.Name)
        return name == "halloweendoor" or string.find(name, "halloweendoor", 1, true) ~= nil
    end

    local function applyInstantPrompt(prompt)
        if not prompt or not prompt:IsA("ProximityPrompt") then return end
        pcall(function()
            if State.OriginalHoldDurations[prompt] == nil then
                State.OriginalHoldDurations[prompt] = prompt.HoldDuration
            end
            if Settings.InstantPP then
                prompt.HoldDuration = 0
                prompt.RequiresLineOfSight = false
                if Settings.BoostReach and prompt.MaxActivationDistance < Settings.MaxReachDistance then
                    prompt.MaxActivationDistance = Settings.MaxReachDistance
                end
            end
        end)
    end

    local function setInstantPPState(enabled)
        Settings.InstantPP = enabled
        if enabled then
            for _, obj in ipairs(Workspace:GetDescendants()) do
                if obj:IsA("ProximityPrompt") then applyInstantPrompt(obj) end
            end
            table.insert(State.PPConnections, Workspace.DescendantAdded:Connect(function(desc)
                if desc:IsA("ProximityPrompt") then task.defer(applyInstantPrompt, desc) end
            end))
            table.insert(State.PPConnections, ProximityPromptService.PromptButtonHoldBegan:Connect(function(prompt)
                if not Settings.InstantPP then return end
                pcall(function()
                    prompt.HoldDuration = 0
                    prompt.RequiresLineOfSight = false
                    if fireproximityprompt then fireproximityprompt(prompt) end
                end)
            end))
        else
            for _, c in ipairs(State.PPConnections) do pcall(function() c:Disconnect() end) end
            table.clear(State.PPConnections)
        end
    end

    local function setDoorESPState(enabled)
        Settings.DoorESP = enabled
        if not enabled then
            for door in pairs(State.HighlightedDoors) do
                if door then
                    local hl = door:FindFirstChild("HalloweenDoor_Highlight")
                    if hl then pcall(hl.Destroy, hl) end
                end
            end
            table.clear(State.HighlightedDoors)
            return
        end
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if isDoorModel(obj) and not State.HighlightedDoors[obj] then
                State.HighlightedDoors[obj] = true
                pcall(function()
                    local hl = Instance.new("Highlight", obj)
                    hl.Name = "HalloweenDoor_Highlight"
                    hl.Adornee = obj
                    hl.FillColor = Color3.fromRGB(255, 115, 0)
                    hl.FillTransparency = 0.5
                    hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
                end)
            end
        end
    end

    local function getNearestDoor()
        local root = getRootPart()
        if not root then return nil end
        local nearestDoor, shortestDist = nil, math.huge
        for _, obj in ipairs(Workspace:GetDescendants()) do
            if isDoorModel(obj) then
                local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)))
                if part then
                    local d = (part.Position - root.Position).Magnitude
                    if d < shortestDist then
                        shortestDist = d
                        nearestDoor = obj
                    end
                end
            end
        end
        return nearestDoor
    end

    local function teleportToNearestDoor()
        local door = getNearestDoor()
        local root = getRootPart()
        if not door or not root then return end
        local part = door:IsA("BasePart") and door or (door:IsA("Model") and (door.PrimaryPart or door:FindFirstChildWhichIsA("BasePart", true)))
        if part then
            root.CFrame = CFrame.lookAt(part.Position + (part.CFrame.LookVector * 4), part.Position)
        end
    end

    local function interactNearestDoor()
        local door = getNearestDoor()
        if not door then return end
        local prompt = door:FindFirstChildWhichIsA("ProximityPrompt", true)
        if prompt and fireproximityprompt then
            pcall(function()
                prompt.HoldDuration = 0
                prompt.RequiresLineOfSight = false
                fireproximityprompt(prompt)
            end)
        end
    end

    local function getAllDoors()
        if State.IsGrabbingAll then return end
        State.IsGrabbingAll = true
        task.spawn(function()
            local root = getRootPart()
            if root then
                for _, obj in ipairs(Workspace:GetDescendants()) do
                    if isDoorModel(obj) then
                        local part = obj:IsA("BasePart") and obj or (obj:IsA("Model") and (obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart", true)))
                        local prompt = obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                        if part and prompt and root.Parent then
                            root.CFrame = part.CFrame + Vector3.new(0, 1, 0)
                            for _ = 1, 3 do
                                pcall(function()
                                    prompt.HoldDuration = 0
                                    if fireproximityprompt then fireproximityprompt(prompt, 0) end
                                end)
                                task.wait(0.05)
                            end
                            task.wait(0.15)
                        end
                    end
                end
            end
            State.IsGrabbingAll = false
        end)
    end

    EventTab:CreateSection("Doors")
    EventTab:CreateToggle({ Name = "Door ESP", CurrentValue = false, Callback = setDoorESPState })
    EventTab:CreateButton({ Name = "Get All Doors", Callback = getAllDoors })
    EventTab:CreateButton({ Name = "TP to Door", Callback = teleportToNearestDoor })
    EventTab:CreateButton({ Name = "Interact Door", Callback = interactNearestDoor })
    EventTab:CreateSection("Prompts")
    EventTab:CreateToggle({ Name = "Instant Prompts", CurrentValue = false, Callback = setInstantPPState })
    EventTab:CreateToggle({ Name = "Boost Reach", CurrentValue = false, Callback = function(v) Settings.BoostReach = v if Settings.InstantPP then setInstantPPState(true) end end })
    EventTab:CreateSlider({ Name = "Reach Distance", Range = {10, 60}, Increment = 1, Suffix = " studs", CurrentValue = 25, Callback = function(v) Settings.MaxReachDistance = v if Settings.InstantPP and Settings.BoostReach then setInstantPPState(true) end end })
end

            ["Left Shoulder"] = torso:FindFirstChild("Left Shoulder"),
            ["Right Shoulder"] = torso:FindFirstChild("Right Shoulder"),
        }
    end

    local function stopSealTech()
        if state.animRenderConnection then
            state.animRenderConnection:Disconnect()
            state.animRenderConnection = nil
        end
        local joints = getCharacterJoints()
        if joints then
            for jName, j in pairs(joints) do
                if j and state.cachedDefaultC0s[jName] then j.C0 = state.cachedDefaultC0s[jName] end
            end
        end
        table.clear(state.cachedDefaultC0s)
    end

    local function startSealTech()
        stopSealTech()
        local joints = getCharacterJoints()
        if not joints then return end
        for name, j in pairs(joints) do
            if j then state.cachedDefaultC0s[name] = j.C0 end
        end
        state.animRenderConnection = RunService.RenderStepped:Connect(function()
            local cur = getCharacterJoints()
            if not cur then stopSealTech() return end
            if cur["Left Shoulder"] and state.cachedDefaultC0s["Left Shoulder"] then
                cur["Left Shoulder"].C0 = state.cachedDefaultC0s["Left Shoulder"] * CFrame.new(0, -58.68, 0)
            end
            if cur["Right Shoulder"] and state.cachedDefaultC0s["Right Shoulder"] then
                cur["Right Shoulder"].C0 = state.cachedDefaultC0s["Right Shoulder"] * CFrame.new(0, -69.16, 0)
            end
        end)
    end

    local function unfreezeBoth()
        state.isFrozen = false
        state.savedElevation = nil
        if state.activeAnimTrack then
            pcall(function() state.activeAnimTrack:AdjustSpeed(1) end)
            state.activeAnimTrack = nil
        end
        if state.frozenTargetChar then
            local tHRP = state.frozenTargetChar:FindFirstChild("HumanoidRootPart")
            if tHRP then tHRP.Anchored = false end
        end
        state.frozenTargetChar = nil
    end

    track(RunService.Heartbeat:Connect(function()
        if not state.infStunEnabled then return end
        local myChar = LocalPlayer.Character
        local hum = myChar and myChar:FindFirstChildOfClass("Humanoid")
        local animator = hum and hum:FindFirstChildOfClass("Animator")
        if animator and not state.isFrozen then
            for _, trackObj in ipairs(animator:GetPlayingAnimationTracks()) do
                local animId = trackObj.Animation and trackObj.Animation.AnimationId or ""
                if animId:find("108418554887656") and trackObj.TimePosition >= 1.81 then
                    trackObj:AdjustSpeed(0)
                    state.activeAnimTrack = trackObj
                    local myHRP = myChar:FindFirstChild("HumanoidRootPart")
                    if myHRP then state.savedElevation = myHRP.Position.Y end
                    state.isFrozen = true
                    trackObj.Stopped:Once(function()
                        if state.isFrozen then unfreezeBoth() end
                    end)
                end
            end
        end
        if state.isFrozen then
            local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
            if myHRP and state.savedElevation and myHRP.Position.Y < state.savedElevation then
                myHRP.CFrame = CFrame.new(myHRP.Position.X, state.savedElevation, myHRP.Position.Z) * myHRP.CFrame.Rotation
            end
        end
    end))

    YutaTab:CreateSection("Yuta Moveset & Tech")
    YutaTab:CreateToggle({ Name = "Insta Seal/kill Tech", CurrentValue = false, Callback = function(v) state.sealTechEnabled = v if v then startSealTech() else stopSealTech() end end })
    YutaTab:CreateToggle({ Name = "Inf Stun (Second Wind)", CurrentValue = false, Callback = function(v) state.infStunEnabled = v if not v then unfreezeBoth() end end })
end

        LockedTarget = nil,
        WasOrbiting = false,
        LastM1Time = 0,
        NoclipParts = {},
        LastDodgeTime = 0,
        IsDodging = false
    }

    local function getHRP(model)
        if not model then return nil end
        return model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso") or model:FindFirstChild("UpperTorso")
    end

    local function getAliveHumanoid(model)
        if not model then return nil end
        local hum = model:FindFirstChildOfClass("Humanoid")
        return (hum and hum.Health > 0) and hum or nil
    end

    local function isTargetValidAndAlive(model)
        if not model or not model.Parent or not model:IsDescendantOf(Workspace) then return false end
        if model:GetAttribute("Dead") then return false end
        if model:FindFirstChildOfClass("ForceField") then return false end
        local hum = getAliveHumanoid(model)
        local hrp = getHRP(model)
        if not (hum and hrp) then return false end
        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        if myHRP and (hrp.Position - myHRP.Position).Magnitude > Settings.MaxTargetRange then
            return false
        end
        return true
    end

    local function restoreCameraSubject()
        local myChar = LocalPlayer.Character
        local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
        if Camera and myHum and Camera.CameraSubject ~= myHum then
            Camera.CameraSubject = myHum
        end
    end

    local function ResetVelocity(hrp)
        if hrp and hrp.Parent then
            pcall(function()
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end
    end

    local function fireOrbitM1(targetChar)
        local now = os.clock()
        if now - State.LastM1Time < 0.08 then return end
        State.LastM1Time = now

        task.spawn(function()
            local myChar = LocalPlayer.Character
            if not myChar then return end
            local fired = false
            local ms = myChar:GetAttribute("Moveset") or LocalPlayer:GetAttribute("Moveset")

                local svc = services and services:FindFirstChild(cleanName .. "Service")

                if act then
                    pcall(function() act:FireServer(false, targetChar) end)
                    fired = true
                end
            end

                if fallback and fallback:IsA("RemoteEvent") then
                    pcall(function() fallback:FireServer(false) end)

        end)
    end

    local function getPlayersList()

        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer then table.insert(list, p.Name) end
        end
        return list
    end

    local function findNewClosestTarget()
        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        if not myHRP then return nil end
        local maxRangeSq = Settings.MaxTargetRange ^ 2
        local myPos = myHRP.Position

        if Settings.TargetMode == "Selected" then
            local pName = Settings.SelectedPlayerName
            if pName and pName ~= "None" then
                local targetPlayer = Players:FindFirstChild(pName)
                if targetPlayer and targetPlayer.Character and isTargetValidAndAlive(targetPlayer.Character) then
                    return targetPlayer.Character
                end
            end
            return nil
        end

        local best, shortest = nil, maxRangeSq
        for _, p in ipairs(Players:GetPlayers()) do
            if p ~= LocalPlayer and p.Character and isTargetValidAndAlive(p.Character) then
                local tHRP = getHRP(p.Character)
                local dSq = (tHRP.Position - myPos).Magnitude ^ 2
                if dSq < shortest then
                    shortest = dSq
                    best = p.Character
                end
            end
        end

        if not best then
            local charsFolder = Workspace:FindFirstChild("Characters") or Workspace:FindFirstChild("Live")
            local pool = charsFolder and charsFolder:GetChildren() or Workspace:GetChildren()
            for _, obj in ipairs(pool) do
                if obj:IsA("Model") and obj ~= myChar and not Players:GetPlayerFromCharacter(obj) and isTargetValidAndAlive(obj) then
                    local tHRP = getHRP(obj)
                    local dSq = (tHRP.Position - myPos).Magnitude ^ 2
                    if dSq < shortest then
                        shortest = dSq
                        best = obj
                    end
                end
            end
        end
        return best
    end

    local function getLockedOrClosestTarget()
        if State.LockedTarget and isTargetValidAndAlive(State.LockedTarget) then
            return State.LockedTarget
        end
        State.LockedTarget = findNewClosestTarget()
        return State.LockedTarget
    end

    local function stepOrbitFrame(deltaTime)
        local anyActive = Settings.OrbitEnabled or Settings.LockBehindEnabled
        if not anyActive then
            if State.WasOrbiting then
                State.WasOrbiting = false
                State.LockedTarget = nil
                restoreCameraSubject()
                local myChar = LocalPlayer.Character
                local myHum = myChar and myChar:FindFirstChildOfClass("Humanoid")
                if myHum then myHum.AutoRotate = true end
            end
            return
        end

        local myChar = LocalPlayer.Character
        local myHRP = getHRP(myChar)
        local myHum = getAliveHumanoid(myChar)
        if not myHRP or not myHum then
            restoreCameraSubject()
            return
        end

        local targetChar = getLockedOrClosestTarget()
        local tHRP = getHRP(targetChar)
        local tHum = getAliveHumanoid(targetChar)
        if not (tHRP and tHum) then
            restoreCameraSubject()
            if not myHum.AutoRotate then myHum.AutoRotate = true end
            return
        end

        State.WasOrbiting = true

        if (Settings.AutoNoclip or Settings.OrbitEnabled) and myChar then
            for _, part in ipairs(myChar:GetDescendants()) do
                if part:IsA("BasePart") and part.CanCollide then part.CanCollide = false end
            end
        end

        if Settings.LockBehindEnabled then
            restoreCameraSubject()
            myHum.AutoRotate = false
            local tPos = tHRP.Position
            local tLook = Vector3.new(tHRP.CFrame.LookVector.X, 0, tHRP.CFrame.LookVector.Z)
            tLook = (tLook.Magnitude > 0.01) and tLook.Unit or -Vector3.zAxis
            local leadOffset = Settings.PredictMovement and (Vector3.new(tHRP.AssemblyLinearVelocity.X, 0, tHRP.AssemblyLinearVelocity.Z) * math.clamp(deltaTime * 1.5, 0.01, 0.1)) or Vector3.zero
            local desiredPos = tPos - (tLook * Settings.LockBehindDistance) + Vector3.new(0, Settings.LockBehindHeight, 0) + leadOffset
            local goalCFrame = CFrame.lookAt(desiredPos, Vector3.new(tPos.X, desiredPos.Y, tPos.Z))
            myHRP.CFrame = Settings.SmoothTracking and myHRP.CFrame:Lerp(goalCFrame, math.clamp(Settings.TrackingLerpSpeed, 0.05, 1)) or goalCFrame
            ResetVelocity(myHRP)
        elseif Settings.OrbitEnabled then
            myHum.AutoRotate = false
            if Camera and Camera.CameraSubject ~= tHum then
                Camera.CameraSubject = tHum
            end

            local stepRad = math.rad(Settings.OrbitSpeed) * deltaTime
            if Settings.OrbitSpeed >= 2500 then
                stepRad = stepRad + 2.39996322972865332
            end
            State.CurrentAngle = (State.CurrentAngle + stepRad) % (math.pi * 2)

            local tPos = tHRP.Position
            local calculatedPos = tPos + Vector3.new(
                math.cos(State.CurrentAngle) * Settings.OrbitDistance,
                Settings.OrbitHeight,
                math.sin(State.CurrentAngle) * Settings.OrbitDistance
            )
            myHRP.CFrame = CFrame.lookAt(calculatedPos, Vector3.new(tPos.X, calculatedPos.Y, tPos.Z))
            ResetVelocity(myHRP)

            if Settings.OrbitAutoM1 then
                fireOrbitM1(targetChar)
            end
        end
    end

    track(RunService.PreSimulation:Connect(stepOrbitFrame))
    track(RunService.RenderStepped:Connect(function(dt)
        if Settings.OrbitEnabled and Settings.OrbitSpeed >= 2500 then
            stepOrbitFrame(dt)
        end
    end))

    track(LocalPlayer.CharacterAdded:Connect(function()
        State.LockedTarget = nil
    end))

    OPTab:CreateToggle({

    OPTab:CreateToggle({

        end
    })
    OPTab:CreateSlider({ Name = "Orbit Speed", Range = {360, 25000}, Increment = 100, Suffix = " °/s", CurrentValue = 7200, Callback = function(val) Settings.OrbitSpeed = val end })
    OPTab:CreateSlider({ Name = "Orbit Distance", Range = {2, 25}, Increment = 0.5, Suffix = " Studs", CurrentValue = 6.5, Callback = function(val) Settings.OrbitDistance = val end })
    OPTab:CreateSlider({ Name = "Orbit Height", Range = {-5, 15}, Increment = 0.5, Suffix = " Studs", CurrentValue = 0, Callback = function(val) Settings.OrbitHeight = val end })
    OPTab:CreateSection("Lock Behind (Back Hunter)")

    OPTab:CreateSlider({ Name = "Height Offset", Range = {-5, 10}, Increment = 0.5, Suffix = " Studs", CurrentValue = 0, Callback = function(val) Settings.LockBehindHeight = val end })
    OPTab:CreateToggle({ Name = "Predict Movement", CurrentValue = false, Callback = function(val) Settings.PredictMovement = val end })
    OPTab:CreateToggle({ Name = "Smooth Tracking", CurrentValue = false, Callback = function(val) Settings.SmoothTracking = val end })

        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local rj = hrp and hrp:FindFirstChild("RootJoint")
        if rj then return rj end
        local lt = char and char:FindFirstChild("LowerTorso")
        return lt and lt:FindFirstChild("Root") or nil
    end

    local function stopInvisibility()
        InvisState.Active = false
        if InvisState.StepConn then InvisState.StepConn:Disconnect() InvisState.StepConn = nil end
        if InvisState.FakePart then pcall(InvisState.FakePart.Destroy, InvisState.FakePart) InvisState.FakePart = nil end
        local char = LocalPlayer.Character
        if char then
            local rj = InvisState.RootJoint or getRootJoint(char)
            if rj and InvisState.OriginalC0 then rj.C0 = InvisState.OriginalC0 end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum.AutoRotate = true
                Camera.CameraSubject = hum
            end
        end
        InvisState.OriginalC0 = nil
        InvisState.RootJoint = nil
    end

    local function startInvisibility(char)
        if not char then return false end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not (hum and hrp) then return false end
        stopInvisibility()
        InvisState.Active = true
        InvisState.RootJoint = getRootJoint(char)
        if InvisState.RootJoint then
            InvisState.OriginalC0 = InvisState.RootJoint.C0
            InvisState.RootJoint.C0 = InvisState.OriginalC0 - Vector3.new(0, INVIS_OFFSET_Y, 0)
        end
        hum.AutoRotate = false
        local fake = Instance.new("Part")
        fake.Name = "JJSFakeTorso"
        fake.Size = Vector3.new(2, 2, 1)
        fake.Transparency = 1
        fake.CanCollide = false
        fake.Anchored = true
        fake.Parent = Workspace
        InvisState.FakePart = fake
        Camera.CameraType = Enum.CameraType.Custom
        Camera.CameraSubject = fake

        InvisState.StepConn = RunService.RenderStepped:Connect(function()
            if not InvisState.Active or not hrp.Parent then return end
            local rj = InvisState.RootJoint
            if rj and rj.Parent and InvisState.OriginalC0 then
                rj.C0 = InvisState.OriginalC0 - Vector3.new(0, INVIS_OFFSET_Y, 0)
            end
            local camLook = Camera.CFrame.LookVector
            local flatLook = Vector3.new(camLook.X, 0, camLook.Z)
            if flatLook.Magnitude > 0.01 then
                hrp.CFrame = CFrame.new(hrp.Position, hrp.Position + flatLook.Unit)
            end
            if InvisState.FakePart then
                InvisState.FakePart.CFrame = hrp.CFrame * CFrame.new(0, 1.5, 0)
            end
        end)
        return true
    end

local function hashStr(str)
    local h = 7
    for i = 1, #str do h = (h * 31 + str:byte(i)) % 1000000007 end
    return tostring(h)
end

local function applyUrl(url, fileName)
    if not fsOk then return end
    task.spawn(function()
        local path = BG_DIR .. "/" .. fileName
        if not isfile(path) then
            local data = httpGet(url)
            if not data or #data < 200 then return end
            if not pcall(writefile, path, data) then return end
        end
        local ok, asset = pcall(function() return getAsset(path) end)
        if ok and asset and asset ~= "" then
            BgImage.Image = asset
        end
    end)
end

local function applyPreset(p)
    applyUrl(p.Url, p.File)
end

local function randomAnime()
    task.spawn(function()
        local body = httpGet("https://wallhaven.cc/api/v1/search?q=anime&categories=010&purity=100&sorting=random&atleast=1280x720&ratios=16x9")
        local ok, json = pcall(function() return HttpService:JSONDecode(body) end)
        if not (ok and type(json) == "table" and type(json.data) == "table" and #json.data > 0) then return end
        local item = json.data[math.random(#json.data)]
        local url = (item.thumbs and item.thumbs.original) or item.path
        if url then applyUrl(url, "bg_rand_" .. tostring(item.id) .. ".jpg") end
    end)
end

local userLinks = {}
local function addPresetFromUrl(url)
    for _, p in ipairs(BG_PRESETS) do
        if p.Url == url then return p end
    end
    local p = { Name = "Link #" .. (#BG_PRESETS + 1), File = "bg_url_" .. hashStr(url) .. ".jpg", Url = url }
    table.insert(BG_PRESETS, p)
    table.insert(userLinks, url)
    return p
end

if fsOk and type(readfile) == "function" then
    pcall(function()
        if isfile(LINKS_FILE) then
            local list = HttpService:JSONDecode(readfile(LINKS_FILE))
            for _, u in ipairs(list) do if type(u) == "string" then addPresetFromUrl(u) end end
        end
    end)
end

local function saveLinks()
    if fsOk then pcall(writefile, LINKS_FILE, HttpService:JSONEncode(userLinks)) end
end
